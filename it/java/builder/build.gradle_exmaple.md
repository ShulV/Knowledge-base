```gradle
// Подключаем плагины — они расширяют возможности Gradle
plugins {
id 'java'         // Добавляет поддержку Java: можно компилировать, тестировать и собирать проект
id 'application'  // Позволяет запускать программу командой ./gradlew run
}

// Настраиваем версию Java
java {
toolchain {
// Указываем, какую версию Java использовать при сборке проекта
// Даже если у тебя установлена другая версия, Gradle сам подберёт нужную
languageVersion = JavaLanguageVersion.of(17)
}
}

// Основная информация о проекте
group = 'org.example'      // Название группы (обычно совпадает с доменом компании)
version = '1.0.0'          // Версия проекта (нужна, если ты собираешь библиотеку)

// Репозитории — места, откуда Gradle будет скачивать зависимости
repositories {
mavenCentral() // Основной публичный репозиторий с библиотеками Java
}

// Зависимости проекта — внешние библиотеки, которые нужны твоему коду
dependencies {
// implementation — библиотека, используемая в основном коде
implementation 'com.google.guava:guava:33.0.0-jre'

    // testImplementation — библиотека, которая нужна только для тестов
    testImplementation 'org.junit.jupiter:junit-jupiter:5.10.0'
}

// Настройка тестов
test {
useJUnitPlatform() // Подключаем JUnit 5 (фреймворк для тестов)
}
```
